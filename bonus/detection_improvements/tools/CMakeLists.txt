cmake_minimum_required(VERSION 3.10)
project(ft_detection_benchmark LANGUAGES CXX)
find_package(catkin REQUIRED COMPONENTS ft_detector)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs calib3d)
find_package(yaml-cpp REQUIRED)
get_filename_component(PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_path(ORT_INCLUDE onnxruntime_cxx_api.h HINTS "${PROJECT_ROOT}/.deps/onnxruntime/include" REQUIRED)
find_library(ORT_LIBRARY onnxruntime HINTS "${PROJECT_ROOT}/.deps/onnxruntime/lib" REQUIRED)
add_executable(benchmark_nn benchmark_nn.cpp)
target_compile_features(benchmark_nn PRIVATE cxx_std_17)
target_compile_options(benchmark_nn PRIVATE -Wall -Wextra)
target_include_directories(benchmark_nn PRIVATE ${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
target_include_directories(benchmark_nn SYSTEM PRIVATE ${ORT_INCLUDE})
target_link_libraries(benchmark_nn ${catkin_LIBRARIES} ${OpenCV_LIBRARIES} yaml-cpp ${ORT_LIBRARY})
